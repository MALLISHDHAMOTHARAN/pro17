
#!/bin/bash

echo "Enter a number:"
read num

flag=0

for ((i=2; i<=num/2; i++))
do
    if [ $((num%i)) -eq 0 ]; then
        flag=1
        break
    fi
done

if [ $flag -eq 0 ]; then
    echo "Prime Number"
else
    echo "Not a Prime Number"
fi

Checks if a user exists in the system.
#!/bin/bash

echo "Enter username:"
read user

if id "$user" &>/dev/null; then
    echo "User exists"
else
    echo "User does not exist"
fi
